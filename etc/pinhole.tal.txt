@circle/<create> ( r* segments* -- )
	,&n STR2
	,&r STR2
	LIT2r 0000
	&>l
		STH2kr /point #0000 STH2kr INC2 DUP2 ,&n LDR2 NEQ2 ?{ POP2 #0000 }
		/point #0000 scene/<push-edge>
		INC2r STH2kr ;&n LDA2 NEQ2 ?/>l
	POP2r JMP2r

@circle/point ( i* -- x* y* )
	#0648 MUL2 [ LIT2 &n $2 ] DIV2 DUP2 x16/cos
	( r ) [ LIT2 &r $2 ] STH2k x16/mul SWP2 x16/sin STH2r !x16/mul

@arc/<create> ( r* segments* angle* -- )
	,&a STR2
	,&n STR2
	,&r STR2
	LIT2r 0000
	&>l
		STH2kr /point #0000 STH2kr INC2 /point #0000 scene/<push-edge>
		INC2r STH2kr ,&n LDR2 NEQ2 ?/>l
	POP2r JMP2r

@arc/point ( i* -- x* y* )
	[ LIT2 &a $2 ] MUL2 [ LIT2 &n $2 ] DIV2 DUP2 x16/cos
	( | )
	[ LIT2 &r $2 ] STH2k x16/mul SWP2 x16/sin STH2r !x16/mul

@cube/<create> ( size* -- )
	,&s STR2
	#00
	&>a
		#01
		&>b
			DUP2 AND ?{
				DUP2 ORA /corner STH2
				STH2
				STH2
				OVR /corner STH2r STH2r STH2r scene/<push-edge> }
			DUP ADD DUP #08 NEQ ?/>b
		POP INC DUP #08 NEQ ?/>a
	POP JMP2r

@cube/corner ( i -- x* y* z* )
	( x ) STHk #01 AND /sign
	( y ) STHkr #02 AND /sign
	( z ) STHr #04 AND
	( >> )

@cube/sign ( bit -- s* )
	[ LIT2 &s $2 ] ROT ?{ #0000 SWP2 SUB2 }
	JMP2r

@viewport/<set> ( x* y* width* height* -- )
	OVR2 .&width STZ2
	OVR2 #01 SFT2 .&width-half STZ2
	DUP2 .&height STZ2
	DUP2 #01 SFT2 .&height-half STZ2
	GTH2k ?/width-greater
	x16/from-u16-rem SWP2 STH2
	DIV2 x16/from-s16 STH2r !{
	&width-greater SWP2 x16/from-u16-rem SWP2 STH2
	DIV2 x16/from-s16 STH2r SWP2 }
	x16/div #0001 x16/add .&a STZ2
	.&y STZ2
	.&x STZ2
	JMP2r

@viewport/<set-frustum> ( near* far* -- )
	.&far STZ2
	.&near STZ2
	JMP2r

	( ) LIT2 &ay $2 .&f LDZ2 x16/mul ,&az LDR2 x16/div DUP2 #0100 inside STHr AND STH
	( ) LIT2 &bx $2 .&f LDZ2 x16/mul LIT2 &bz $2 x16/div DUP2 .&a LDZ2 inside STHr AND STH
	( ) LIT2 &by $2 .&f LDZ2 x16/mul ,&bz LDR2 x16/div DUP2 #0100 inside STHr AND ?{

@viewport/point-to-viewport-clamping ( point** -- point'** )

	( | y )
	.&height-half LDZ2 x16/mul .&height-half LDZ2 ADD2 #0000
	( ) .&height LDZ2 #0001 SUB2 x16/clamp .&y LDZ2 ADD2 SWP2
	( | x )
	.&height-half LDZ2 x16/mul .&width-half LDZ2 ADD2
	( ) #0000 .&width LDZ2 #0001 SUB2 x16/clamp .&x LDZ2 ADD2 SWP2 JMP2r

	,&y2 STR2
	,&x2 STR2
	STH2
	STH2
	( | x )
	[ LIT2 ADD2r SUB2r ] ,&x2 LDR2 STH2kr SUB2k s16/abs ,&dx STR2
	s16/gth [ JMP SWP POP ] ,&sx STR
	SWP2r
	( | y )
	[ LIT2 ADD2r SUB2r ] ,&y2 LDR2 STH2kr SUB2k s16/abs #0000 SWP2 SUB2 ,&dy STR2
	s16/gth [ JMP SWP POP ] ,&sy STR
	,&dx LDR2 ,&dy LDR2 ADD2
	&>w

			DUP2r [ LITr -Screen/y ] DEO2r
			OVR2r [ LITr -Screen/x ] DEO2r

				( x ) OVR2r STH2r [ LIT2 &x2 $2 ] NEQ2 ?{
					( stop ) POP2 POP2r POP2r JMP2r } }
			( e -> e2 ) DUP2k ADD2 DUP2
			( y ) [ LIT2 &dy $2 ] s16/lth ?{
				( e+dy ) SWP2 ,&dy LDR2 ADD2 SWP2
				( x1+sx ) SWP2r [ LIT2r 0001 ] [ &sx $1 ] SWP2r }
			( x ) [ LIT2 &dx $2 ] s16/gth ?/>w2

		( y1+sy ) [ LIT2r 0001 ] [ &sy $1 ] !/>w

@crash ( -> )
	#01 DEO
	#010f BRK

@get-direction ( a*** b*** -- dir** )
	INC2 ;&bz STA2
	;&by STA2
	;&bx STA2
	INC2 ;&az STA2
	;&ay STA2
	;&ax STA2
	[ LITr 08 ]
	&>try
		;&ax LDA2 ;&az LDA2 x16/div x16/is-inf ?/next
		;&ay LDA2 ;&az LDA2 x16/div x16/is-inf ?/next
		;&bx LDA2 ;&bz LDA2 x16/div x16/is-inf ?/next
		;&by LDA2 ;&bz LDA2 x16/div x16/is-inf ?/next
		!/end
		&next STHkr #00 EQU ?/end
		;&az LDA2k #0200 x16/mul SWP2 STA2
		;&bz LDA2k #0200 x16/mul SWP2 STA2
		LITr 01 SUBr !/>try
	&end POPr [ LIT2r =&az ] LDA2r ;&ax LDA2k STH2kr x16/div SWP2 STA2
	;&ay LDA2k STH2r x16/div SWP2 STA2
	[ LIT2r =&bz ] LDA2r ;&bx LDA2k STH2kr x16/div SWP2 STA2
	;&by LDA2k STH2r x16/div SWP2 STA2
	;&b DUP2 ;&a OVR2 vector/<sub>
	#0000 ;&bz STA2
	DUP2 vector/<normalize>
	;&bx LDA2 ;&by LDA2 JMP2r
	&a &ax $2 &ay $2 &az $2 &b &bx $2 &by $2 &bz $2

@clip-line-planes ( a*** b*** -- unclipped )
	;&bz STA2
	;&by STA2
	;&bx STA2
	;&az STA2
	;&ay STA2
	;&ax STA2
	;&az LDA2 /compute-outcode ;&bz LDA2 /compute-outcode ORAk ?{
		POP2 ;&ax LDA2 ;&ay LDA2 ;&az LDA2 ;&bx LDA2 ;&by LDA2 ;&bz LDA2 #01 JMP2r }
	ANDk #00 EQU ?{ POP2 #00 JMP2r }
	( | Failed both tests. )
	DUP #00 EQU ?{
		DUP /inflate-outcode STH2k ;&bz LDA2 x16/sub ;&az LDA2 ;&bz LDA2 x16/sub x16/div STH2
		;&bx LDA2 ;&ax LDA2 OVR2 x16/sub STH2kr x16/mul x16/add ;&bx STA2
		;&by LDA2 ;&ay LDA2 OVR2 x16/sub STH2r x16/mul x16/add ;&by STA2
		LIT2r =&bz STA2r }
	POP DUP #00 EQU ?{
		DUP /inflate-outcode STH2k ;&az LDA2 x16/sub ;&bz LDA2 ;&az LDA2 x16/sub x16/div STH2
		;&ax LDA2 ;&bx LDA2 OVR2 x16/sub STH2kr x16/mul x16/add ;&ax STA2
		;&ay LDA2 ;&by LDA2 OVR2 x16/sub STH2r x16/mul x16/add ;&ay STA2
		LIT2r =&az STA2r }
	POP ;&ax LDA2 ;&ay LDA2 ;&az LDA2 ;&bx LDA2 ;&by LDA2 ;&bz LDA2 #01 JMP2r
	&ax $2 &ay $2 &az $2 &bx $2 &by $2 &bz $2

@clip-line-planes/compute-outcode ( z* -- outcode )
	OVR #80 AND #00 EQU ?{ POP2 .Outcode/near JMP2r }
	DUP2 .viewport/near LDZ2 LTH2 #00 EQU ?{ POP2 .Outcode/near JMP2r }
	DUP2 .viewport/far LDZ2 LTH2 ?{ POP2 .Outcode/far JMP2r }
	POP2 .Outcode/center JMP2r

@clip-line-planes/inflate-outcode ( outcode -- d* )
	DUP .Outcode/near NEQ ?{ POP .viewport/near LDZ2 JMP2r }
	DUP .Outcode/far NEQ ?{ POP .viewport/far LDZ2 JMP2r }
	!crash

@clip-to-vp ( a** b** dir** xmin* ymin* xmax* ymax* -- unclipped )
	;&ymax STA2
	;&xmax STA2
	;&ymin STA2
	;&xmin STA2
	;&diry STA2
	;&dirx STA2
	;&by STA2
	;&bx STA2
	;&ay STA2
	;&ax STA2
	;&ax LDA2 ;&ay LDA2 /compute-outcode ;&bx LDA2 ;&by LDA2 /compute-outcode
	&loop ( -- )
	ORAk ?{
		POP2 ;&ax LDA2 ;&ay LDA2 ;&bx LDA2 ;&by LDA2 #01 JMP2r }
	ANDk #00 EQU ?{ POP2 #00 JMP2r }
	( | Failed both tests. )
	OVR ?{ &calculate-b DUP /compute-intersection/from-a NIP !/loop }
	DUP ?{ &calculate-a OVR /compute-intersection/from-b ROT POP SWP !/loop }
	/is-a-inf /is-b-inf ANDk #00 EQU ?{
		( infinity ) POP2 POP2 #00 JMP2r }
	POP ?/calculate-a
	!/calculate-b
	( a ) &a &ax $2 &ay $2
	( b ) &b &bx $2 &by $2
	( d ) &dir &dirx $2 &diry $2

@clip-to-vp/compute-outcode ( point** -- outcode )
	LITr -Outcode/center
	( | handlers )
	DUP2 [ LIT2 &ymin $2 ] x16/cmp INC ?{ LITr -Outcode/bottom ORAr }
	[ LIT2 &ymax $2 ] x16/cmp #01 NEQ ?{ LITr -Outcode/top ORAr }
	DUP2 [ LIT2 &xmin $2 ] x16/cmp INC ?{ LITr -Outcode/left ORAr }
	[ LIT2 &xmax $2 ] x16/cmp #01 NEQ ?{ LITr -Outcode/right ORAr }
	STHr JMP2r

@clip-to-vp/inflate-outcode ( outcode -- axis* d* side )
	DUP .Outcode/top AND #00 EQU ?{
		POP ;Vector/y ;&ymax LDA2 .Side/top JMP2r }
	DUP .Outcode/bottom AND #00 EQU ?{
		POP ;Vector/y ;&ymin LDA2 .Side/bottom JMP2r }
	DUP .Outcode/right AND #00 EQU ?{
		POP ;Vector/x ;&xmax LDA2 .Side/right JMP2r }
	DUP .Outcode/left AND #00 EQU ?{
		POP ;Vector/x ;&xmin LDA2 .Side/left JMP2r }
	!crash

@clip-to-vp/is-a-inf ( -- inf )
	;&ax LDA2 x16/is-inf ?{ ;&ay LDA2 !x16/is-inf }
	#01 JMP2r

@clip-to-vp/is-b-inf ( -- inf )
	;&bx LDA2 x16/is-inf ?{ ;&by LDA2 !x16/is-inf }
	#01 JMP2r

@clip-to-vp/compute-intersection/from-a ( outcode -- outcode' )
	( ) /inflate-outcode STH
	SWP2 STH2
	( ) ;&a STH2kr ADD2 LDA2 x16/sub
	( ) ;&dir STH2r ADD2 LDA2 x16/div STH2
	( ) ;&ax LDA2 ;&dirx LDA2 STH2kr x16/mul x16/add
	( ) ;&ay LDA2 ;&diry LDA2 STH2r x16/mul x16/add
	( ) STHkr /align-point
	( ) ;&ax LDA2 ;&ay LDA2 ;&bx LDA2 ;&by LDA2 STHr
	( ) /clamp-point
	( ) OVR2 OVR2 /compute-outcode STH
	( ) ;&by STA2
	;&bx STA2
	STHr JMP2r

@clip-to-vp/compute-intersection/from-b ( outcode -- outcode' )
	( ) /inflate-outcode STH
	SWP2 STH2
	( ) ;&b STH2kr ADD2 LDA2 SWP2 x16/sub
	( ) ;&dir STH2r ADD2 LDA2 x16/div STH2
	( ) ;&bx LDA2 ;&dirx LDA2 STH2kr x16/mul x16/sub
	( ) ;&by LDA2 ;&diry LDA2 STH2r x16/mul x16/sub
	( ) STHkr /align-point
	( ) ;&bx LDA2 ;&by LDA2 ;&ax LDA2 ;&ay LDA2 STHr
	( ) /clamp-point
	( ) OVR2 OVR2 /compute-outcode STH
	( ) ;&ay STA2
	;&ax STA2
	STHr JMP2r

@clip-to-vp/align-point ( point** side -- point'** )
	DUP .Side/top EQU ?/align-top
	DUP .Side/bottom EQU ?/align-bottom
	DUP .Side/right EQU ?/align-right
	DUP .Side/left EQU ?/align-left
	!crash
	&align-top POP POP2 #0100 JMP2r
	&align-bottom POP POP2 #ff00 JMP2r
	&align-right POP NIP2 .viewport/a LDZ2 SWP2 JMP2r
	&align-left POP NIP2 .viewport/a LDZ2 #0000 SWP2 SUB2 SWP2 JMP2r

@clip-to-vp/clamp-point ( point** target** prev** side -- point'** )
	DUP .Side/top EQU ?/clamp-horizontally
	DUP .Side/bottom EQU ?/clamp-horizontally
	DUP .Side/right EQU ?/clamp-vertically
	DUP .Side/left EQU ?/clamp-vertically

@clip-to-vp/clamp-horizontally ( point** target** prev** side -- point'** )

@x16/constants
|0192 &pi-half
|0324 &pi
|04b6 &3pi-half
|0648 &2pi
|5183 &256-pi
|7fff &maximum
|8000 &err
|8001 &minimum
|/constants

%x16/bias ( x* y* -- x* y* ) {
	#8000 EOR2 SWP2 #8000 EOR2 SWP2 }

%x16/neg ( x* -- -x* ) {
	#0000 SWP2 SUB2 }

@x16/is-non-neg ( x* -- bool )
	POP #80 LTH JMP2r

@x16/is-neg ( x* -- bool )
	POP #7f GTH JMP2r

@x16/cmp ( x* y* -- c )
	/bias
	( >> )

@x16/ucmp ( x* y* -- c )
	LTH2k ?{ GTH2 JMP2r }
	POP2 POP2 #ff JMP2r

@x16/eq ( x* y* -- x=y )
	EQU2 JMP2r

@x16/ne ( x* y* -- x!=0 )
	NEQ2 JMP2r

@x16/lt ( x* y* -- x<y )
	/bias LTH2 JMP2r

@x16/lteq ( x* y* -- x<=y )
	/bias GTH2 #00 EQU JMP2r

@x16/gt ( x* y* -- x>y )
	/bias GTH2 JMP2r

@x16/gteq ( x* y* -- x>=y )
	/bias LTH2 #00 EQU JMP2r

@x16/min ( x* y* -- min[x, y]* )
	OVR2 OVR2 /lt [ JMP SWP2 ] POP2 JMP2r

@x16/clamp ( n* min* max* -- clamp )
	ROT2 /min
	( >> )

@x16/max ( x* y* -- max[x, y]* )
	OVR2 OVR2 /gt [ JMP SWP2 ] POP2 JMP2r

@x16/is-whole ( x* -- bool )
	NIP #00 EQU JMP2r

@x16/is-inf ( n* -- n==inf||n==-inf? )
	DUP2 #7fff EQU2 STH
	#8001 EQU2 STH
	ORAr STHr JMP2r

@x16/remainder ( x* y* -- x%y* )
	OVR #80 LTH ?{ /neg }
	SWP2 OVR #80 LTH STHk ?{ /neg }
	SWP2 DIV2k MUL2 SUB2 STHr ?{ /neg }
	JMP2r

@x16/sub ( x* y* -- x-y* )
	/neg
	( >> )

@x16/add ( x* y* -- x+y* )
	EOR2k POP #80 AND ?{
		OVR2 STH2
		ADD2 DUP2 STH2kr EOR2 POP #80 AND ?{
			POP2r DUP2 #8000 NEQ2 ?{ POP2 #8001 }
			JMP2r }
		POP2 STH2r #7fff GTH2 ?{ #7fff JMP2r }
		#8001 JMP2r }
	ADD2 JMP2r

@x16/negate ( x* -- -x* )
	/neg JMP2r

@x16/recip ( n* -- 1/n* )
	#0100 SWP2
	( >> )

@x16/div ( x* y* -- x/y* )
	;&div-unsigned !/signed-op

@x16/quotient ( x* y* -- x//y* )
	;&quot-unsigned !/signed-op

@x16/mul ( x* y* -- xy* )
	;&mul-unsigned
	( >> )

@x16/signed-op ( x* y* f* -- f[x,y] )
	STH2
	[ LIT2r 0001 ]
	( y ) OVR #80 LTH ?{ /neg SWPr }
	SWP2
	( x ) OVR #80 LTH ?{ /neg SWPr }
	SWP2 SWP2r STH2r JSR2 STHr ?{ /neg }
	POPr JMP2r

@x16/mul-unsigned ( ab* cd* -- ab*cd* )
	DUP ?{ !/mul-whole }
	OVR2 NIP ?{ SWP2 !/mul-whole }
	OVR ?{ !/mul-frac }
	OVR2 POP ?{ SWP2 !/mul-frac }
	ROT STH
	STH2
	#00 SWP #7fff OVR2 DIV2 STH2kr LTH2 ?{
		( ) STH2kr MUL2
		( ) STH2r #00 SWP
		( ) #00 STHkr MUL2 #08 SFT2 ROT #00 SWP #00
		( ) STHr MUL2 ADD2 ADD2 OVR #7f GTH ?/mul-o2
		JMP2r }
	POP2r POPr &mul-o2 POP2 #7fff JMP2r

	&mul-whole ( ab* cd* -- ab*cd* )
	POP DUP ?{ POP2 POP #0000 JMP2r }
	#00 SWP #7fff OVR2 DIV2 ROT2 LTH2k ?{ NIP2 MUL2 JMP2r }
	POP2 POP2 POP2 #7fff JMP2r

	&mul-frac ( ab* cd* -- ab*cd* )
	STH
	#08 SFT2 #00 STHkr MUL2 #08 SFT2 ROT #00 SWP #00 STHr MUL2 ADD2 JMP2r

@x16/normalize-angle ( n* -- n'* )
	;&256-pi
	( >> )

@x16/div-unsigned ( x* y* -- x/y* )
	ORAk ?{
		POP2 ORA ?{ #0000 JMP2r }
		#7fff JMP2r }
	DUP ?{ #08 SFT2 DIV2 JMP2r }
	DIV2k DUP2 #0080 LTH2 ?{
		POP2 POP2 POP2 #7fff JMP2r }
	STHk OVR2 MUL2 ROT2 SWP2 SUB2 OVR2 POP ?{
		NIP #00 SWP2 DIV2 NIP STHr SWP JMP2r }
	OVR2 POP #10 LTH ?{
		LIT2r 0001
		&>lp
			#10 SFT2 DUP2r ADD2r GTH2k ?{ OVR2 SUB2 INC2r }
			STH2kr POP ?{ !/>lp }
		POP2 POP2 STH2r NIP STHr SWP JMP2r }
	#40 SFT2 OVR2 DIV2k STHk MUL2 SUB2 #40 SFT2 OVR2 DIV2k STHk MUL2 SUB2 POP2 POP2
	( ) STHr STHr #40 SFT ORA STHr SWP JMP2r

@x16/quot-unsigned ( x* y* -- x//y* )
	DIV2 DUP2 #007f GTH2 ?{ NIP #00 JMP2r }
	POP2 #7fff JMP2r

@x16/floor ( x* -- floor[x]* )
	DUP EOR JMP2r

@x16/ceil ( x* -- ceil[x]* )
	#00ff ADD2 DUP EOR JMP2r

@x16/round ( x* -- round[x]* )
	OVR #01 AND #7f ADD #00 SWP ADD2 DUP EOR JMP2r

@x16/unit-circle ( x* -- x'* )
	DUP2 ;&2pi LTH2 ?{
		OVR #80 LTH ?{
			/neg /unit-circle ORAk ?{ JMP2r }
			;&2pi SWP2 SUB2 JMP2r }
		DUP2 ;&2pi DIV2 DUP2 #01 SFT2 STH2
		;&2pi MUL2 SUB2 STH2r SUB2 OVR #80 LTH ?{ ;&2pi ADD2 } }
	JMP2r

@x16/cos ( x* -- cos[x]* )
	/unit-circle ;&pi-half ADD2 DUP2 ;&2pi LTH2 ?/sin-reduced
	;&2pi SUB2 !/sin-reduced

@x16/sin ( x* -- sin[x]* )
	OVR #80 LTH ?/sin-positive
	/neg /sin-positive !/negate
	( >> )
	&sin-positive ( x* -- sin[x]* )
	/unit-circle
	( >> )
	&sin-reduced ( x* -- sin[x]* )
	DUP2 ;&3pi-half LTH2 ?{
		;&2pi SWP2 SUB2 /sin-q !/negate }
	DUP2 ;&pi LTH2 ?{ ;&pi SUB2 /sin-q !/negate }
	DUP2 ;&pi-half LTH2 ?{ ;&pi SWP2 SUB2 }
	&sin-q ( x* -- sin[x]* )
	DUP2 #0183 LTH2 ?{ POP2 #0100 JMP2r }
	;&sin-table ADD2 LDA #00 SWP JMP2r

@x16/tan ( x* -- tan[x]* )
	/unit-circle DUP2 ;&3pi-half EQU2 ?/tan-error
	DUP2 ;&pi-half EQU2 ?/tan-error
	DUP2 ;&3pi-half LTH2 ?{
		;&2pi SWP2 SUB2 /tan-q !/negate }
	DUP2 ;&pi LTH2 ?{ ;&pi SUB2 !/tan-q }
	DUP2 ;&pi-half LTH2 ?{
		;&pi SWP2 SUB2 /tan-q !/negate }
	&tan-q ( x* -- tan[x]* )
	DUP2 ADD2 ;&tan-table ADD2 LDA2 JMP2r

	&tan-error ( x* -- tan[x]* )
	POP2 ;&err JMP2r

@x16/atan2 ( y* x* -- angle* )
	DUP2 ;&err EQU2 ?/atan2-error
	OVR2 ;&err EQU2 ?/atan2-error
	SWP2 OVR #80 LTH STHk ?{ /neg }
	SWP2 OVR #80 LTH STHk ?{ /neg }
	LTH2k STHk ?{ SWP2 }
	/div-unsigned #0100 OVR2 SUB2 #0046 /mul-unsigned #00c9 ADD2 /mul-unsigned STHr ?{ ;&pi-half SWP2 SUB2 }
	STHr ?{ ;&pi SWP2 SUB2 }
	STHr ?{ ;&2pi SWP2 SUB2 }
	JMP2r

	&atan2-error ( y* x* -- angle* )
	POP2 POP2 ;&err JMP2r

	POP2 #7fff JMP2r
