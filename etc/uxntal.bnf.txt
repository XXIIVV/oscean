/* Uxntal 54K formal grammar, written in W3C-style EBNF */

Program		::= WS? (Statement WS)*

Statement	::= Opcode 
			  | ParentLabel
			  | ChildLabel 
			  | PadRel 
			  | PadAbs 
			  | LiteralHex 
			  | RawString
			  | ImmJSI 
			  | ImmJMI
			  | ImmJCI
			  | LiteralZep 
			  | LiteralRel 
			  | LiteralAbs 
			  | RawZep 
			  | RawRel
			  | RawAbs
			  | Ignore
			  | Comment
			  | MacroName

Opcode		::= Op Mode*
Op			::= 'BRK' | 'INC' | 'POP' | 'DUP' | 'NIP' | 'SWP' | 'OVR' | 'ROT'
			  | 'EQU' | 'NEQ' | 'GTH' | 'LTH' | 'JMP' | 'JCN' | 'JSR' | 'STH'
			  | 'LDZ' | 'STZ' | 'LDR' | 'STR' | 'LDA' | 'STA' | 'DEI' | 'DEO'
			  | 'ADD' | 'SUB' | 'MUL' | 'DIV' | 'AND' | 'ORA' | 'EOR' | 'SFT'
			  | 'LIT'
Mode		::= '2' | 'k' | 'r'

ParentLabel	::= '@' Name
ChildLabel	::= '&' Sublabel
PadRel		::= '$' (Number | Name | ChildLabel)
PadAbs		::= '|' (Number | Name | ChildLabel)
LiteralHex	::= '#' (Byte | Short)
RawString 	::= '"' (Char - WS)+
ImmJSI		::= Name | AnonLabel
ImmJMI		::= '!' Label
ImmJCI		::= '?' Label
LiteralZep	::= '.' Label
LiteralRel	::= ',' Label
LiteralAbs	::= ';' Label
RawZep		::= '-' Label
RawRel		::= '_' Label
RawAbs		::= '=' Label

Ignore		::= (')' | '[' | ']') Word?

Comment		::= '(' Word? WS (CmtContents WS)* ')'
CmtContents	::= Word - ('(' | ')') | '(' WS (CmtContents WS)* ')'

Macro		::= '%' MacroName WS ((Word - '{') WS)* '{' MacroBody '}'
MacroName	::= Name
MacroBody	::= WS (Statement WS)*

Label		::= Name | ChildLabel | AnonLabel
Name		::= Scope | Scope? '/' Sublabel
Sublabel	::= Word?
AnonLabel	::= '{' Word? WS (Statement WS)* '}' Word?
Scope		::= (ScopeChar - Rune) ScopeChar*
ScopeChar	::= Char - (WS | '/')

Number		::= Nibble Nibble? Nibble? Nibble?
Short		::= Byte Byte
Byte		::= Nibble Nibble
Nibble		::= [0-9a-f]
Rune		::= '|' | '$' | '@' | '&' | ',' | '_' | '.' | '-'
		 	  | ';' | '=' | '!' | '?' | '#' | '"' | '~' | '%'
			  | '{' | '}' | '(' | ')' | '[' | ']'
Word		::= (Char - WS)+
WS			::= (#x20 | #x9 | #xD | #xA)+
