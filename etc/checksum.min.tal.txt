( 54K . Checksum/min )

|10 @Console &vector $2 &read $1 &pad $4 &type $1 &write $1 &error $1

|100

@on-reset ( -> )
	;hash/on-console .Console/vector DEO2
	BRK

@hash/on-console ( -> )
	[ LIT2 04 -Console/type ] DEI NEQ ?{
		,&x LDR2 ,&y LDR2 u32/<print>
		#0a18 DEO
		BRK }
	.Console/read DEI DUP DUP2
	( x ) [ LIT2 &x 1234 ] #2443 MUL2 ADD2 ,&x STR2
	( y ) [ LIT2 &y abcd ] #118d MUL2 ADD2 ,&y STR2
	BRK

@u32/<print> ( hs* ls* -- )
	SWP2 /u16
	&u16 ( hb lb -- )
	SWP /u8
	&u8 ( xx -- )
	DUP #04 SFT /u4
	&u4 ( x -- )
	#0f AND DUP #09 GTH #27 MUL ADD #30 ADD .Console/write DEO
	JMP2r
