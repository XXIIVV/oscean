@src/on-console ( -> )
	.Console/read DEI DUP #20 GTH ?{
		POP <print-program>
		#800f DEO
		BRK }
	[ LIT2 00 &ptr -&buf ] INCk ,&ptr STR
	STZ2

@src/<print-name> ( -- )
	,&ptr LDR
	( | rewind )
	&>wr
		LDZk [ LIT "/ ] EQU ?{ #01 SUB LDZk ?/>wr }
	INC
	( | forward )
	&>w
		LDZk [ LIT ". ] EQU ?{
			LDZk #18 DEO
			INC LDZk ?/>w }
	POP JMP2r

	[ LIT2r 0000 ]
	&>stream
	
		&>continue

				;dict/c !str/<print> }

			[ LIT &buf $1 ] u8/<print>

			INC2r

		#0a18 DEO
		#0918 DEO !/>stream
