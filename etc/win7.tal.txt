@path/<init> ( -- )
	.&buf LDZ ?{ ;dict/home-path !/<set> }
	;&buf str/cap ,&ptr STR
	POP JMP2r

@path/make-parent ( -- str* )
	/get-depth #02 GTH ?{ ;dict/home JMP2r }
	[ LIT2 00 _&ptr ] LDR /up /up INC !decstr/make-dir

@path/make-location ( -- str* )
	/get-depth #01 GTH ?{ ;dict/home JMP2r }
	[ LIT2 00 _&ptr ] LDR /up INC !decstr/make-dir

@path/get-depth ( -- depth )
	[ LITr 00 ] [ LIT2 -&buf _&ptr ] LDR
	&>rgd
		#01 SUB LDZk [ LIT "/ ] NEQ ?{ INCr }
		NEQk ?/>rgd
	POP2 STHr JMP2r

@path/<set> ( path* -- )
	[ LIT2 -&buf _&ptr ] STR
	( >> )

@path/<push-str> ( subpath* -- )
	&>wpu
		LDAk DUP ?{ POP POP2 JMP2r }
		/<push>
		INC2 !/>wpu

@path/<push> ( c -- )
	[ LIT2 00 &ptr -&buf ]
	( | clamp )
	.&cap LTHk ?{
		( | TODO: Warn overflow )
		SWP }
	POP
	( ) INCk ,&ptr STR
	STZ2
	JMP2r

@path/pop ( -- )
	;&buf ;dict/home-path str/cmp ?{
		,&ptr LDR /up INC #00 OVR STZ
		,&ptr STR }
	JMP2r

@path/up ( ptr -- addr )
	#01 SUB .&buf SWP
	&>rup
		#01 SUB LDZk [ LIT "/ ] EQU ?{ NEQk ?/>rup }
	NIP JMP2r

@path/<reload> ( -- )
	;&buf .File/name DEO2
	;&memlen .File/length DEO2
	;&mem .File/read DEO2
	directory/<reset>
	/<nullws>
	/<parse-folders> !/<parse-files>

@path/<parse-folders> ( -- )
	;&mem
	&>wfo
		LDAk ?{ POP2 JMP2r }
		LDAk LIT "- NEQ ?{ directory/push }
		#0005 ADD2 str/cap INC2 !/>wfo

@path/<parse-files> ( -- )
	;&mem
	&>wfi
		LDAk ?{ POP2 JMP2r }
		LDAk LIT "- EQU ?{ directory/push }
		#0005 ADD2 str/cap INC2 !/>wfi

@path/<nullws> ( -- )
	;&mem
	&>wnw
		LDAk DUP ?{ POP POP2 JMP2r }
		#1f GTH ?{ LITr 00 STH2k STAr }
		INC2 !/>wnw

@path/<print> ( -- )
	;&buf !str/<print>

@path/<reqload> ( -- )

	header/<reqdraw> !window/<bind-screen>

@path/<delay-load> ( -- )

	( | is active )
	INC DUP ?{
		( | is complete )
		/<reload>
		window/<reqdraw-all> }
	,&timer STR !window/<bind-screen>

@directory/<bind> ( -- )
	;&on-button .Controller/vector DEO2
	;&on-mouse .Mouse/vector DEO2
	JMP2r

@directory/on-button ( -> )
	.Controller/button DEI2 DUP2 manifest/scan INC2k ORA ?{ POP2 !/on-key }
	( * ) JSR2 POP2 BRK

@directory/on-key ( button key -> )
	sel/get-length ORA ?{
		NIP DUP ?{ POP BRK }
		sel/<reset>
		search/<bind-put>
		BRK }
	NIP DUP ?{ POP BRK }
	command/<bind-put>
	BRK

@directory/on-mouse ( -> )
	cursor/<update>
	( | check scroll )
	.Mouse/sy DEI2 ORAk ?{
		POP2
		( | no scroll )
		[ LIT2 &last $1 -Mouse/state ] DEI DUP ,&last STR
		[ LIT2 02 -Controller/button ] DEI EQU ?/on-mouse-alt
		( | get id )
		DUP2 #0001 EQU2 ?/on-mouse-touch
		DUP2 #0002 EQU2 ?/on-mouse-click
		POP2
		( | Touch )
		BRK }
	scrollbar/<mod>
	BRK

@directory/on-mouse-touch ( last state -> )
	POP2 .Mouse/x DEI2 .Mouse/y DEI2 file/pos-id /<pick-id>
	BRK

@directory/on-mouse-click ( last state -> )
	POP2 .Mouse/x DEI2 .Mouse/y DEI2 file/pos-id /<pick-id>
	/<open>
	BRK

@directory/on-mouse-alt ( last state -> )
	ORAk ?{ POP2 BRK }
	#0001 EQU2 ?{
		( | on drag )
		( x ) .Mouse/x DEI2 [ LIT2 &ax $2 ] SUB2 [ LIT2 &w $2 ] ADD2 #33 SFT2
		( y ) .Mouse/y DEI2 [ LIT2 &ay $2 ] SUB2 [ LIT2 &h $2 ] ADD2 #33 SFT2 window/<set-size>
		BRK }
	( | on down )
	.Mouse/x DEI2 ,&ax STR2
	.Mouse/y DEI2 ,&ay STR2
	window/get-width ,&w STR2
	window/get-height ,&h STR2
	BRK
