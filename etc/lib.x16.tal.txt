( x16 )

@x16/constants
|0192 &pi-half
|0324 &pi
|04b6 &3pi-half
|0648 &2pi
|7fff &maximum
|8000 &err
|8001 &minimum
|/constants %uxnfor-fix { }

%x16/bias ( x* y* -- x* y* ) {
	#8000 EOR2 SWP2 #8000 EOR2 SWP2 }

%x16/mul8 ( x y -- xy* ) {
	#0000 SWP2 ROT SWP MUL2 }

@x16/is-non-neg ( x* -- bool )
	;&minimum LTH2 JMP2r

@x16/is-neg ( x* -- bool )
	;&maximum GTH2 JMP2r

@x16/cmp ( x* y* -- c )
	/bias
	( >> )

@x16/ucmp ( x* y* -- c )
	LTH2k ?{ GTH2 JMP2r }
	POP2 POP2 #ff JMP2r

@x16/eq ( x* y* -- x=y )
	EQU2 JMP2r

@x16/ne ( x* y* -- x!=0 )
	NEQ2 JMP2r

@x16/lt ( x* y* -- x<y )
	/bias LTH2 JMP2r

@x16/lteq ( x* y* -- x<y )
	/bias GTH2 #00 EQU JMP2r

@x16/gt ( x* y* -- x<y )
	/bias GTH2 JMP2r

@x16/gteq ( x* y* -- x<y )
	/bias LTH2 #00 EQU JMP2r

@x16/min ( x* y* -- min[x, y]* )
	OVR2 OVR2 /lt [ JMP SWP2 ] POP2 JMP2r

@x16/max ( x* y* -- max[x, y]* )
	OVR2 OVR2 /gt [ JMP SWP2 ] POP2 JMP2r

@x16/is-whole ( x* -- bool )
	NIP #00 EQU JMP2r

@x16/remainder ( x* y* -- x%y* )
	OVR2 OVR2 /quotient /mul
	( >> )

@x16/sub ( x* y* -- x-y* )
	/negate
	( >> )

@x16/add ( x* y* -- x+y* )
	EOR2k POP #80 AND ?{
		OVR2 STH2
		ADD2 DUP2 STH2kr EOR2 POP #80 AND ?{
			POP2r DUP2 #8000 NEQ2 ?{ POP2 #8001 }
			JMP2r }
		POP2 STH2r #7fff GTH2 ?{ #7fff JMP2r }
		#8001 JMP2r }
	ADD2 JMP2r

@x16/negate ( x* -- -x* )
	#0000 SWP2 SUB2 JMP2r

@x16/div ( x* y* -- x/y* )
	;&div-unsigned !/signed-op

@x16/quotient ( x* y* -- x//y* )
	;&quot-unsigned !/signed-op

@x16/mul ( x* y* -- xy* )
	;x16-mul-unsigned
	( >> )

@x16/signed-op ( x* y* f* -- f[x,y] )
	STH2
	[ LIT2r 0001 ] DUP2 #8000 LTH2 ?{ /negate SWPr }
	SWP2 DUP2 #8000 LTH2 ?{ /negate SWPr }
	SWP2 SWP2r STH2r JSR2 STHr ?{ /negate }
	POPr JMP2r

@x16-mul-unsigned ( ab* cd* -- ac+ad+bc+bd* )
	DUP ?{ !/mul-whole }
	OVR2 NIP ?{ SWP2 !/mul-whole }
	OVR ?{ !/mul-frac }
	OVR2 POP ?{ SWP2 !/mul-frac }
	OVR2 OVR2 STH2
	STH2
	ROT SWPr ROTr SWP2 x16/mul8 DUP2 #007f GTH2 ?/mul-o1
	SWP SWP2 x16/mul8 #08 SFT2 ADD2 STH2r x16/mul8 ADD2 OVR #7f GTH ?/mul-o3
	STH2r x16/mul8 ADD2 OVR #7f GTH ?/mul-o4
	JMP2r
	&mul-o1 POP2 POP2r &mul-o3 POP2r &mul-o4 POP2 #7fff JMP2r
	&mul-whole POP STH
	SWP STHkr x16/mul8 DUP2 #007f GTH2 ?{
		SWP ROT STHr x16/mul8 ADD2 DUP2 #8000 LTH2 ?{ POP2 #7fff }
		JMP2r }
	POP2 POP POPr #7fff JMP2r
	&mul-frac STH
	POP STHkr x16/mul8 #08 SFT2 ROT STHr x16/mul8 ADD2 JMP2r

@x16/div-unsigned ( x* y* -- x/y* )
	ORAk ?{
		POP2 ORA ?{ #0000 JMP2r }
		#7fff JMP2r }
	DUP ?{
		POP #00 SWP DIV2 JMP2r }
	DIV2k DUP2 #007f GTH2 ?/div-o1
	STH2k LITr 80 SFT2r OVR2 STH2
	MUL2 SUB2 STH2r [ LIT2r 0100 ]
	&>la1
		&la2 STH2kr #0000 EQU2 ?/div-done
		OVR2 #7fff GTH2 ?{
			SWP2 #10 SFT2 SWP2 LITr 01 SFT2r LTH2k ?/la2
			SWP2 OVR2 SUB2 SWP2 DUP2r ROT2r ADD2r SWP2r !/>la1 }
	&>lb1
	
		&>lb2
			STH2kr #0000 EQU2 ?/div-done
			#01 SFT2 LITr 01 SFT2r LTH2k ?/>lb2
		SWP2 OVR2 SUB2 SWP2 DUP2r ROT2r ADD2r SWP2r !/>lb1
	&div-done POP2 POP2 POP2r STH2r DUP2 #7fff GTH2 ?/div-o2
	JMP2r
	&div-o1 POP2 POP2 &div-o2 POP2 #7fff JMP2r

@x16/quot-unsigned ( x* y* -- x//y* )
	DIV2 DUP2 #007f GTH2 ?{ #80 SFT2 JMP2r }
	POP2 #7fff JMP2r

@x16/floor ( x* -- floor[x]* )
	DUP EOR JMP2r

@x16/ceil ( x* -- floor[x]* )
	#00ff ADD2 DUP EOR JMP2r

@x16/round ( x* -- round[x]* )
	OVR #01 AND ?{
		#007f ADD2 DUP EOR JMP2r }
	#0080 ADD2 DUP EOR JMP2r

@x16/unit-circle ( x* -- x'* )
	DUP2 ;&2pi LTH2 ?{
		DUP2 #8000 LTH2 ?{
			/negate /unit-circle ORAk #00 EQU ?{ ;&2pi SWP2 SUB2 }
			JMP2r }
		DUP2 ;&2pi DIV2 DUP2 #01 SFT2 STH2
		;&2pi MUL2 SUB2 STH2r SUB2 DUP2 #8000 LTH2 ?{ ;&2pi ADD2 } }
	JMP2r

@x16/cos ( x* -- cos[x]* )
	/unit-circle ;&pi-half ADD2 DUP2 ;&2pi LTH2 ?/sin-reduced
	;&2pi SUB2 !/sin-reduced

@x16/sin ( x* -- sin[x]* )
	DUP2 ;&err LTH2 ?/sin-positive
	/negate /sin-positive !/negate
	( >> )
	&sin-positive ( x* -- sin[x]* )
	/unit-circle
	( >> )
	&sin-reduced ( x* -- sin[x]* )
	DUP2 ;&3pi-half LTH2 ?{
		;&2pi SWP2 SUB2 /sin-q !/negate }
	DUP2 ;&pi LTH2 ?{ ;&pi SUB2 /sin-q !/negate }
	DUP2 ;&pi-half LTH2 ?{ ;&pi SWP2 SUB2 }
	&sin-q ( x* -- sin[x]* )
	DUP2 #0183 LTH2 ?{ POP2 #0100 JMP2r }
	;&sin-table ADD2 LDA #00 SWP JMP2r

@x16/tan ( x* -- tan[x]* )
	/unit-circle DUP2 ;&3pi-half EQU2 ?/tan-error
	DUP2 ;&pi-half EQU2 ?/tan-error
	DUP2 ;&3pi-half LTH2 ?{
		;&2pi SWP2 SUB2 /tan-q !/negate }
	DUP2 ;&pi LTH2 ?{ ;&pi SUB2 !/tan-q }
	DUP2 ;&pi-half LTH2 ?{
		;&pi SWP2 SUB2 /tan-q !/negate }
	&tan-q ( x* -- tan[x]* )
	DUP2 ADD2 ;&tan-table ADD2 LDA2 JMP2r

	&tan-error ( x* -- tan[x]* )
	POP2 ;&err JMP2r

@x16/atan2 ( y* x* -- angle* )
	DUP2 ;&err EQU2 ?/atan2-error
	OVR2 ;&err EQU2 ?/atan2-error
	( ) OVR2 /is-neg #00 SWP ROT2 ROT2
	( ) DUP2 /is-neg #00 SWP ROT2 ROT2 DUP2 #8000 LTH2 ?{ /negate }
	SWP2 DUP2 #8000 LTH2 ?{ /negate }
	SWP2 OVR2 OVR2 /gt DUP STH
	#00 EQU ?{ SWP2 }
	/div STHr ROT ROT #0100 OVR2 /sub #0046 /mul #00c9 /add /mul ROT #00 EQU ?{ ;&pi-half SWP2 /sub }
	SWP2 NIP #00 EQU ?{ ;&pi SWP2 /sub }
	SWP2 NIP #00 EQU ?{ ;&2pi SWP2 /sub }
	JMP2r

	&atan2-error ( y* x* -- angle* )
	POP2 POP2 ;&err JMP2r

@x16/from-s8 ( n -- x* )
	#00 JMP2r

@x16/from-s16 ( n* -- x* )
	DUP2 #ff80 GTH2 ?/from-s16-neg
	DUP2 #007f GTH2 ?/from-s16-error &from-s16-neg NIP #00 JMP2r
	&from-s16-error POP2 ;&err JMP2r

@x16/to-s16 ( x* -- whole* )
	DUP2 #7fff GTH2 ?{ DUP EOR SWP JMP2r }
	#00ff STHk ADD2 POP DUP #07 SFT STHr MUL SWP JMP2r

@x16/to-s8 ( x* -- whole )
	DUP2 #0f SFT2 #00ff MUL2 ADD2 POP JMP2r

@x16/<init-sqrt> ( -- )
	#00
	&>l
		#00 OVR /sqrt #00ff /min NIP OVR #00 SWP ;&sqrt-table ADD2 STA
		INC DUP ?/>l
	POP JMP2r

@x16/sqrt ( x* -- sqrt[x]* )
	ORAk ?{ JMP2r }
	DUP2 #0020
	&>sqrt-l1
		OVR2 #0004 LTH2 ?{
			SWP2 #02 SFT2 SWP2 #10 SFT2 !/>sqrt-l1 }
	NIP2
	&>sqrt-l2
		OVR2 OVR2 /div OVR2 ADD2 #01 SFT2 GTH2k ?{ POP2 NIP2 JMP2r }
		NIP2 !/>sqrt-l2

@x16/sq ( x* -- x*x* )
	DUP2 #8000 LTH2 ?{ #0000 SWP2 SUB2 }
	DUP2 #0b50 GTH2 ?{
		STH2k #08 SFT2 DUP2 MUL2 #80 SFT2 STH2kr #08 SFT2 STH2kr #00ff AND2 MUL2 #10 SFT2 ADD2 STH2r #00ff AND2 DUP2 MUL2 #08 SFT2 ADD2 JMP2r }
	POP2 #7fff JMP2r

@x16/sqrt-fast ( x* -- sqrt[x]* )
	LITr 00
	&>norm
		OVR #00 EQU ?{ #02 SFT2 INCr !/>norm }
	NIP #00 SWP ;&sqrt-table ADD2 LDA #00 SWP STHr #40 SFT SFT2 JMP2r

@x16/<emit> ( x* -- )
	DUP2 #8000 LTH2 ?{
		#2d18 DEO
		#0000 SWP2 SUB2 }
	SWP DUP #64 LTH ?/<100
	#64 DIVk DUP /<emit-dec-digit>
	MUL SUB !/>=10
	&<100 DUP #0a LTH ?/<10
	&>=10
		#0a DIVk DUP /<emit-dec-digit>
		MUL SUB &<10 /<emit-dec-digit>
		#2e18 DEO
		#00 SWP DUP2 #0027 MUL2 SWP2 #0008 ADD2 #04 SFT2 ADD2 #03e8 DIV2k DUP /<emit-dec-digit>
		MUL2 SUB2 #0064 DIV2k DUP /<emit-dec-digit>
		MUL2 SUB2 #000a DIV2k DUP /<emit-dec-digit>
		MUL2 SUB2 NIP
		( >> )

@x16/<emit-dec-digit> ( d -- )
	#30 ADD #18 DEO
	JMP2r
