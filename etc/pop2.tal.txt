	stack/<reset>
	expr/<reset>
	;&on-console .Console/vector DEO2
	;&a !str/<print>

	.Console/read DEI
	( end ) DUP LIT "; NEQ ?{
		POP /<parse>
		/<emit-bindings>
		#0a18 DEO
		token/<bind>
		BRK }
	expr/<push-any>
	BRK

@function/<parse> ( -- )
	token/<reset>
	;expr/buf str/walk-ws str/print-word #2018 DEO
	;&c str/<print>
	&>w
		LDAk DUP ?{
			POP POP2 /<parse-token>
			;&d !str/<print> }
		DUP #20 GTH ?{ POP /<parse-token> !/resume }
		token/<push> &resume INC2 !/>w

	&c "!{ 20 00
	&d "} 20 00

@function/<parse-token> ( -- )
	;token/buf LDAk ?{ POP2 JMP2r }

	pad-prefab/<emit> !token/<reset>

@function/<emit-bindings> ( -- )
	&>web
		stack/depth ?{ JMP2r }
		stack/pop binding-prefab/<emit> !/>web

@end/<bind> ( str* -- )

	;&a str/<print-lb> !token/<bind>

	&a "} 00

@expr/walk-value ( str* -- str* )
	token/<reset>
	LDAk ?{ JMP2r }
	LDAk LIT "# NEQ ?{ LIT "# token/<push>
		INC2 }
	DUP2 /is-goto ?/walk-goto
	DUP2 /is-function ?/walk-function
	DUP2 /is-position ?/walk-position
	&>wv
		LDAk DUP chr/is-an ?{ POP !/<parse-value> }
		token/<push>
		INC2 !/>wv

	;&nil !stack/<push>

@expr/walk-position ( str* -- str* )

	;token/buf dict/alloc stack/<push>
	;&nil !stack/<push>

	&d 20 "ADD2 20 "LDA2 00

		INC2 !/>wwf

@chr/hex ( c -- val )
	( dec ) [ LIT "0 ] SUB DUP #09 GTH [ JMP JMP2r ]
	( hex ) #27 SUB DUP #0a SUB #05 GTH [ JMP JMP2r ]
	( nil ) POP #ff JMP2r

@chr/is-num ( c -- t )

@chr/is-opr ( c -- t )
	DUP LIT "+ EQU ?/t-opr
	DUP LIT "- EQU ?/t-opr
	DUP LIT "* EQU ?/t-opr
	DUP LIT "/ EQU ?/t-opr
	DUP LIT "% EQU ?/t-opr
	DUP LIT "= EQU ?/t-opr
	DUP LIT "! EQU ?/t-opr
	DUP LIT "< EQU ?/t-opr
	DUP LIT "> EQU ?/t-opr
	DUP LIT "& EQU ?/t-opr
	DUP LIT "| EQU ?/t-opr
	DUP LIT "^ EQU ?/t-opr
	DUP LIT ": EQU ?/t-opr
	POP #00 JMP2r
	&t-opr POP #01 JMP2r

@chr/is-lc ( c -- lowercase )

@chr/is-uc ( c -- uppercase )
