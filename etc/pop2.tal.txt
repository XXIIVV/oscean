@expr/<parse> ( -- )
	;&buf /walk-value stack/pop /<parse-postfix>
	&>wp
		LDAk ?{
			POP2 stack/pop /<parse-postfix>
			chr/<ws> !token/<bind> }
		/walk-operand
		( | parens )
		LDAk LIT "{ NEQ ?{ INC2 }
		LDAk LIT "} NEQ ?{ INC2 ;dict/lda2 str/<print-ws> }
		LDAk LIT "( NEQ ?{ INC2 ;&nil stack/<push> }
		LDAk LIT ") NEQ ?{
			INC2 stack/pop /<parse-postfix>
			stack/pop /<parse-postfix> }
		LDAk LIT ", NEQ ?{ INC2 ;&nil stack/<push> }
		/walk-value stack/pop /<parse-postfix> !/>wp

@expr/walk-value ( str* -- str* )
	token/<reset>
	LDAk ?{ JMP2r }
	DUP2 /is-goto ?/walk-goto
	DUP2 /is-function ?/walk-function
	token/push-label !/<parse-value>

@expr/walk-function ( str* -- str* )
	token/<reset>
	token/push-label dict/alloc stack/<push>
	;&nil !stack/<push>

@expr/is-function ( str* -- t )
	LDAk chr/is-label ?{ POP2 #01 JMP2r }
	/walk-label LDA LIT "( EQU JMP2r

@expr/walk-label ( str* -- str* )
	&>wwa
		LDAk chr/is-label ?{ JMP2r }
		INC2 !/>wwa

@expr/is-goto ( str* -- t )
	LDA2k LIT2 "go EQU2 STH
	INC2 INC2 LDA2 LIT2 "to EQU2 STHr AND JMP2r

@expr/walk-goto ( str* -- str* )
	token/<reset>
	#0004 ADD2 token/push-label !prefab-jump/<emit>

@expr/walk-operand ( str* -- str* )
	token/<reset>
	&>wo
		LDAk DUP chr/is-opr ?{
			POP ;token/buf LDAk ?{ POP2 JMP2r }
			dict/alloc !stack/<push> }
		token/<push>
		INC2 !/>wo

@expr/<reset> ( -- )

	JMP2r

@expr/<push-str> ( str* -- )

@expr/<parse-value> ( str* -- )
	( | handlers )
	DUP2 str/is-dec ?/<parse-value-dec>
	DUP2 str/is-hex ?/<parse-value-hex>
	( | raw value )
	LDAk LIT "# NEQ ?{ INC2 !prefab-refrel/<emit> }
	LDAk LIT "@ NEQ ?{ INC2 !prefab-refabs/<emit> }
	prefab-refrel/<emit>
	stack/is-allocator ?{ ;dict/lda2 !str/<print-ws> }
	JMP2r

@expr/<parse-value-dec> ( str* -- )
	str/dec !prefab-lithex/<emit>

@expr/<parse-value-hex> ( str* -- )

@expr/<parse-postfix> ( str* -- )
	LDAk ?{ POP2 JMP2r }
	LDA2k LIT2 "-> NEQ2 ?{ POP2 ;dict/sta2 !str/<print-ws> }
	LDA2k LIT2 "=> NEQ2 ?{ POP2 ;dict/deo2 !str/<print-ws> }
	LDA2k LIT2 "<- NEQ2 ?{ POP2 ;dict/jsr2 !str/<print-ws> }
	LDAk operators/find INC2k ORA ?{
		POP2
		( call ) ;dict/fn-prefix str/<print> !str/<print-ws> }
	INC2 LDA2 NIP2 !str/<print-ws>

	&nil 00

@chr/hex ( c -- val )
	( dec ) [ LIT "0 ] SUB DUP #09 GTH [ JMP JMP2r ]
	( hex ) #27 SUB DUP #0a SUB #05 GTH [ JMP JMP2r ]
	( nil ) POP #ff JMP2r

@chr/not-sc ( c -- c t )
	DUP LIT "; NEQ JMP2r

@chr/is-num ( c -- t )
	LIT "0 SUB #0a LTH JMP2r

@chr/is-opr ( c -- t )
	operators/find INC2 ORA JMP2r

@chr/is-lc ( c -- lowercase )
	LIT "A SUB #1a LTH JMP2r

@chr/is-uc ( c -- uppercase )
	LIT "a SUB #1a LTH JMP2r

@chr/is-label ( c -- t )
	DUP LIT "_ EQU ?/pt
	DUP LIT "# EQU ?/pt
	DUP LIT "@ EQU ?/pt
	DUP LIT ". EQU ?/pt
	DUP /is-num ?/pt
	DUP /is-lc ?/pt
	DUP /is-uc ?/pt
	POP #00 JMP2r
	&pt POP #01 JMP2r

		INC2 LDAk chr/is-dec ?{ POP2 #00 JMP2r } }
	&>wid

		INC2 !/>wid

@str/is-hex ( str* -- t )

@str/hex ( str* -- value* )
	[ LIT2r 0000 ]
	&>wh
		[ LITr 40 ] SFT2r LDAk chr/hex [ LITr 00 ] STH
		ADD2r INC2 LDAk ?/>wh
	POP2 STH2r JMP2r

@str/dec ( str* -- val* )
	[ LIT2r 0000 ] !{

	&>ldec
		[ LITr 00 ] STH
		MUL2r [ LITr 00 ] STH
		ADD2r INC2 }
	( str* . val* ) LDAk [ LIT "0 ] SUB #0a LTHk ?/>ldec
	POP2 POP2 STH2r JMP2r
