@expr/<parse> ( -- )
	;&buf
	&>wp
		str/walk-ws LDAk ?{ &end POP2 !token/<bind> }
		LDAk LIT "; EQU ?/end
		/walk-fragment !/>wp

@expr/walk-fragment ( str* -- str* )
	str/walk-ws LDAk ?{ JMP2r }
	( | Main )
	LDAk chr/is-label ?/walk-label
	LDAk chr/is-operand ?/walk-operator
	( | Special )
	LDAk LIT "{ NEQ ?{
		INC2 ;dict/backarrow stack/<push>
		;&nil !stack/<push> }
	LDAk LIT "} NEQ ?{ INC2 stack/pop !/<parse-postfix> }
	LDAk LIT "( NEQ ?{ INC2 ;&nil !stack/<push> }
	LDAk LIT ") NEQ ?{
		INC2 stack/pop /<parse-postfix>
		stack/pop !/<parse-postfix> }
	LDAk LIT "" EQU ?/walk-string
	( | Unknown )
	;dict/err-unknown-expr err/<print>
	DUP2 err/<print-lb>
	INC2 JMP2r

@expr/walk-label ( str* -- str* )
	/walk-label-core
	( | Post label )
	stack/pop !/<parse-postfix>

@expr/walk-label-core ( str* -- str* )
	str/walk-ws LDAk ?{ JMP2r }
	token/<reset>
	token/push-label
	( | handlers )
	DUP2 ;dict/goto str/cmp ?/<parse-goto>
	DUP2 str/is-dec ?/<parse-value-dec>
	DUP2 str/is-hex ?/<parse-value-hex>
	LDAk LIT "# NEQ ?{ INC2 !prefab-refrel/<emit> }
	LDAk LIT "@ NEQ ?{ INC2 !prefab-refabs/<emit> }
	( | function )
	OVR2 LDA LIT "( NEQ ?{ dict/alloc stack/<push>
		;&nil !stack/<push> }
	prefab-refrel/<emit>
	stack/is-allocator ?{ ;dict/lda2 !str/<print-ws> }
	JMP2r

@expr/<parse-goto> ( str* token* -- str* )
	( pop goto ) POP2
	( walk to label ) str/walk-ws
	( write jump ) token/<reset>
	token/push-label !prefab-jump/<emit>

@expr/walk-string ( str* -- str* )
	LIT2 "{ 18 DEO
	chr/<ws>
	prefab-string/emit LIT2 "} 18 DEO
	chr/<ws>
	INC2 ;dict/sth2r str/<print-ws>
	stack/pop !/<parse-postfix>

@expr/walk-operator ( str* -- str* )
	str/walk-ws LDAk ?{ JMP2r }
	token/<reset>
	token/push-operand dict/alloc !stack/<push>

@expr/<parse-value-dec> ( str* -- )
	str/dec !prefab-lithex/<emit>

@expr/<parse-value-hex> ( str* -- )
	INC2 INC2 str/hex !prefab-lithex/<emit>

@expr/<parse-postfix> ( str* -- )
	LDAk ?{ POP2 JMP2r }
	LDA2k LIT2 "-> NEQ2 ?{ POP2 ;dict/sta2 !str/<print-ws> }
	LDA2k LIT2 "<- NEQ2 ?{ POP2 ;dict/lda2 !str/<print-ws> }
	LDA2k LIT2 "=> NEQ2 ?{ POP2 ;dict/deo2 !str/<print-ws> }
	LDA2k LIT2 "<> NEQ2 ?{ POP2 ;dict/jsr2 !str/<print-ws> }
	LDAk operators/find INC2k ORA ?{ POP2 !prefab-scope/<emit> }
	INC2 LDA2 NIP2 !str/<print-ws>

	&nil 00

@expr/<reset> ( -- )

	JMP2r

@expr/<push-str> ( str* -- )

@expr/<push> ( c -- )

@expr/push-until-keyword ( c -- reached )

@expr/realloc-keyword ( -- dict* )

@expr/get-last-word ( -- str* )
	;&buf ,&ptr LDR2
	&>l
		NEQ2k ?{ POP2 JMP2r }
		#0001 SUB2 LDAk #20 GTH ?/>l
	NIP2 INC2 JMP2r

@chr/hex ( c -- val )
	( dec ) [ LIT "0 ] SUB DUP #09 GTH [ JMP JMP2r ]
	( hex ) #27 SUB DUP #0a SUB #05 GTH [ JMP JMP2r ]
	( nil ) POP #ff JMP2r

	&>ldec
		[ LITr 00 ] STH
		MUL2r [ LITr 00 ] STH
		ADD2r INC2 }
	( str* . val* ) LDAk [ LIT "0 ] SUB #0a LTHk ?/>ldec
	POP2 POP2 STH2r JMP2r

@keywords/find ( token* -- ffff|addr* )
	STH2
	;&lutend ;&lut
	&>l
		LDA2k STH2kr str/cmp ?{
			#0004 ADD2 GTH2k ?/>l
		POP2 #ffff }
	NIP2 POP2r JMP2r

@keywords/lut [4
	=dict/function =function/<bind> =dict/end =end/<bind>
	=dict/vars =vars/<bind> =dict/comment =comment/<bind>
	=dict/goto =goto/<bind> =dict/if =if/<bind>
	=dict/then =then/<bind> =dict/elseif =elseif/<bind>
	=dict/else =else/<bind> =dict/close =close/<bind>

@operators/find ( c -- ffff|addr* )

	;&lutend ;&lut
	&>l
		LDAk STHkr EQU ?{
			#0003 ADD2 GTH2k ?/>l
		POP2 #ffff }
	NIP2 POPr JMP2r

@operators/lut [4
	"+ =dict/add2 "- =dict/sub2
	"* =dict/mul2 "/ =dict/div2
	"% =dict/mod2 "| =dict/ora2
	"& =dict/and2 "^ =dict/eor2
	"~ =dict/jsr2 "= =dict/equ2-s
	"! =dict/neq2-s "> =dict/gth2-s
	"< =dict/lth2-s ", =&nil
	&lutend ]
	&nil 00
