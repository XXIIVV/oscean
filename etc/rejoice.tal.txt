@token/on-console-interactive ( -> )
	.Console/read DEI DUP /<append>
	#0a NEQ ?{ world/<eval> }
	BRK

@token/on-console ( -> )
	.Console/read DEI
	( | Normalize )
	DUP #20 GTH ?{ POP #20 }
	/<append>
	[ LIT2 04 -Console/type ] DEI NEQ ?{ world/<eval>
		#800f DEO }
	BRK

@token/on-console-comment ( -> )
	[ LIT2 ") -Console/read ] DEI NEQ ?{
		[ LIT2 &comment-callback $2 ] .Console/vector DEO2 }
	BRK

@token/on-console-string ( -> )
	[ LIT2 "" -Console/read ] DEI NEQk ?{
		POP2 [ LIT2 &string-callback $2 ] .Console/vector DEO2
		BRK }
	DUP #20 NEQ ?{
		POP [ LIT "\ ] /<store>
		[ LIT "s ] }
	/<store>
	POP BRK

@token/<append> ( c -- )
	DUP LIT "( NEQ ?{
		.Console/vector DEI2 ,&comment-callback STR2
		;&on-console-comment .Console/vector DEO2
		POP JMP2r }
	DUP LIT "" NEQ ?{
		.Console/vector DEI2 ,&string-callback STR2
		;&on-console-string .Console/vector DEO2
		POP JMP2r }
	DUP LIT "[ NEQ ?{
		,&+depth LDR INC ,&+depth STR }
	DUP LIT "] NEQ ?{
		,&+depth LDR #01 SUB ,&+depth STR }
	[ LIT &+depth $1 ] ?{
		DUP #20 GTH ?{ POP /<parse> !/<reset> } }
	( >> )

@token/<store> ( c -- )
	[ LIT2 00 &+ptr -&+buf ] INCk
	( clamp ) .&+bufend LTHk ?{ ;dict/token err/<set-buf>
		SWP }
	POP ,&+ptr STR
	STZ2
	JMP2r

@token/<reset> ( -- )
	[ LIT2 -&+buf _&+ptr ] STR
	[ LIT2 00 -&+buf ] STZ
	JMP2r

@token/<parse> ( -- )
	.&+buf
	( exists ) LDZk ?{ POP JMP2r }
	( @label ) LDZk LIT "@ EQU ?label/<parse>
	( 'quote ) LDZk LIT "' EQU ?label/<parse-quote>
	world/<push-scope>
	( >> )

@token/<parse-fraction> ( ztr -- )
	( num bag ) item/parse-any-push world/<pop-scope>
	world/<push-scope>
	LDZk LIT "/ EQU ?{
		( den bag [] ) world/<push-identity>
		world/<pop-scope> !/<parse-target> }
	INC
	( den bag ) item/parse-any-push world/<pop-scope> !/<parse-target>

@token/<parse-target> ( ztr -- )
	LDZ2k LIT2 "-> NEQ2 ?{
		( label ) INC INC #00 OVR syms/find-alloc STH2
		symbol/walk /<take-quote>
		POP2 STH2r !/<parse-effect> }
	/<take-quote>
	( >> )

@token/<parse-effect> ( ztr target* -- )
	world/<push>
	LDZ2k LIT2 "=> NEQ2 ?{
		INC INC /<compile-string>
		#00 SWP dict/alloc !world/<push> }
	POP #0000 !world/<push>

@token/<compile-string> ( ztr -- ztr )
	DUP STH
	DUP
	&>wc
		( r w ) OVR LDZ DUP ?{
			POP NIP #00 SWP STZ
			STHr JMP2r }
		DUP LIT "{ NEQ ?{ POP /<interpolate> !/>wc }
	OVR STZ
	INC SWP INC SWP !/>wc

@token/<interpolate> ( r w -- r' w' )
	OVR INC DUP /<find-close>
	LDZk LIT "} NEQ ?{
		#00 OVR STZ
		STH
		#00 SWP syms/find-alloc NIP OVR INC STZ
		#01 OVR STZ
		INC INC NIP STHr INC SWP JMP2r }
	POP2 OVR LDZ OVR STZ
	INC SWP INC SWP JMP2r

@token/<find-close> ( p -- p )
	&>wf
		LDZk ?{ JMP2r }
		LDZk LIT "} NEQ ?{ JMP2r }
		INC !/>wf

@token/<take-quote> ( -- short* )

%tote/get-addr ( id* -- addr* ) {
	DUP2 ADD2 ;tote/buf ADD2 }

%tote/next-cell ( addr* -- next* ) {
	INC2 INC2 }

%tote/has-item ( count* id* -- t ) {
	tote/get-addr LDA2 ADD2 POP #80 AND }

@tote/get-register ( item* -- item* value* )
	INC2k INC2 LDA2 OVR #80 AND ?{ JMP2r }
	syms/mask-id /get-addr LDA2 JMP2r

@tote/apply ( fraction* -- success )
	( | take )
	queue/<reset>
	LDA2k LDA2k SWP2 /next-cell
	&>lac
		DUP2 queue/<take>
		item/next GTH2k ?/>lac
	POP2 POP2
	( | validate )
	queue/can-apply ?{ POP2 #00 JMP2r }
	( | effect )
	DUP2 /<effect>
	( | put )
	LDA2k OVR2 /next-cell
	&>lap
		DUP2 queue/<put>
		item/next GTH2k ?/>lap
	POP2 POP2 POP2 queue/<apply>
	#01 JMP2r

@tote/<effect> ( fraction* -- )
	LDA2 LDA2 INC2 INC2 LDA2 ORAk ?{ POP2 JMP2r }
	str/<set-print>
	str/<print> !str/<set-debug>

@tote/get-range ( -- to* from* )
	( from* ) ;&buf
	( from* to-ptr* ) DUP2 syms/get-length ADD2 ;&bufend
	( from* min* ) LTH2k ?{ SWP2 }
	POP2
	( to* from* ) SWP2 JMP2r

	[ LIT2 "[ 19 ] DEO

	&>lp

			LDA2k #00 STHkr item/<print-count-id>

		/next-cell INCr GTH2k ?/>lp
	POP2 POP2 POP2r [ LIT2 "] 19 ] DEO
	#2019 DEO

@tote/<clear> ( -- )
	/get-range
	&>lc
		#0000 OVR2 STA2
		/next-cell GTH2k ?/>lc
	POP2 POP2 JMP2r

@queue/<take> ( item* -- )
	LDA2k ORAk ?{
		( zero ) POP2
		( id ) POP2 JMP2r }
	( | Handle inf )
	DUP2 tote/get-addr LDA2 ;inf NEQ2 ?{
		OVR2 tote/get-register NIP2 ;inf EQU2 ?{ POP2 POP2 JMP2r } }
	( | Handle dec )
	/find INC2k ORA ?{
		POP2
		( | Not null )
		tote/get-register ORAk ?{ POP2 POP2 JMP2r }
		SWP2
		( id* ) LDA2 /<push>
		( count* ) #0000 SWP2 SUB2 !/<push> }
	( | merge offset )
	INC2 INC2 STH2
	tote/get-register STH2kr LDA2 /sub-sat STH2r STA2
	POP2 JMP2r

@queue/sub-sat ( value* offset* -- new* )
	SWP2 SUB2 DUP2 #8001 LTH2 ?{ JMP2r }
	POP2 #8001 JMP2r

@queue/add-sat ( value* offset* -- new* )
	OVR #80 AND ?{
		ADD2 DUP2 #8000 LTH2 ?{ POP2 #7fff }
		JMP2r }
	ADD2 JMP2r

@queue/<put> ( item* -- )
	LDA2k
	( | Handle inc )
	/find INC2k ORA ?{
		POP2
		( | Not null )
		tote/get-register ORAk ?{ POP2 POP2 JMP2r }
		SWP2
		( id* ) LDA2 /<push>
		( count* ) !/<push> }
	( | merge offset )
	INC2 INC2 STH2
	tote/get-register STH2kr LDA2 /add-sat STH2r STA2
	POP2 JMP2r

@queue/<reset> ( -- )
	;&+buf ,&+ptr STR2
	#0000 ;&+buf STA2
	JMP2r

@queue/<apply> ( -- )
	,&+ptr LDR2 ;&+buf
	&>l
		EQU2k ?{
			INC2k INC2 LDA2 OVR2 LDA2 tote/get-addr LDA2k ROT2 ADD2
			( clamp ) ;inf LTH2k ?{ SWP2 }
			POP2
			( new-value* item* ) SWP2 STA2
			item/next !/>l }
	POP2 POP2 JMP2r

@queue/<push> ( value* -- )
	[ LIT2 &+ptr =&+buf ] INC2k INC2
	( clamp ) ;&+bufend LTH2k ?{ ;dict/queue err/<set-buf>
		SWP2 }
	POP2 ,&+ptr STR2
	STA2
	JMP2r

@queue/find ( item* -- addr* )
	STH2
	,&+ptr LDR2 ;&+buf
	&>lf
		LDA2k STH2kr EQU2 ?{
			item/next GTH2k ?/>lf
		POP2 #ffff }
	NIP2 POP2r JMP2r

@queue/can-apply ( -- t )
	,&+ptr LDR2 ;&+buf
	&>lca

			( item* count* ) INC2k INC2 LDA2
			( item* count* id* ) OVR2 LDA2 tote/has-item ?{ item/next !/>lca } }
	EQU2 JMP2r
