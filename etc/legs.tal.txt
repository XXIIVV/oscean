@world/get-pos ( x y -- isox* isoy* )
	( y ) ADDk STH
	( x ) SUB .&size ADD #00 SWP
	( offsetx ) #40 SFT2
	( offsety ) #00 STHr #30 SFT2 #0080 ADD2 JMP2r

@world/get-shadow ( 0xyz* -- t )
	DUP2 #0f ORA SWP2 INC2
	&>lgs
		NEQ2k ?{ POP2 POP2 #00 JMP2r }
		DUP2 ;&buf ADD2 LDA ?{ INC2 !/>lgs }

	POP2 POP2 JMP2r

@wireworld/<step> ( -- )
	[ LIT2r =&buf ] world/get-bounds
	&>l
		/apply STH2kr STA
		INC2 INC2r GTH2k ?/>l
	POP2 POP2 POP2r
	( | move )
	;&copy-mmu .System/expansion DEO2 !world/<redraw>

	&copy-mmu [ 01 =&buflen 0000 =&buf 0000 =world/buf ]

@wireworld/apply ( addr* -- addr* cell )
	LDAk DUP ?{ JMP2r }
	( . type ) DUP #f0 AND STH
	#0f AND
	( | head )
	DUP #0f NEQ ?{
		POP #0a STHr ORA JMP2r }
	( | tail )
	DUP #0a NEQ ?{
		POP #00 STHr ORA JMP2r }
	( | conductor )
	DUP ?{
		STH
		/has-neighbors ?{ ORAr STHr JMP2r }
		POPr #0f STHr ORA JMP2r }
	( type . ) STHr ORA JMP2r

%wireworld/not-head ( addr* -- addr* t ) {
	LDA #0f AND #0f NEQ }

@wireworld/has-neighbors ( addr* -- addr* t )
	[ LITr ff ]
	( | z-1 )
	( -1,-1,-1 ) DUP2 #feef ADD2 /not-head ?{ INCr }
	( -1, 0,-1 ) DUP2 #feff ADD2 /not-head ?{ INCr }
	( -1,+1,-1 ) DUP2 #ff0f ADD2 /not-head ?{ INCr }
	( .0,-1,-1 ) DUP2 #ffef ADD2 /not-head ?{ INCr }
	( .0, 0,-1 ) DUP2 #ffff ADD2 /not-head ?{ INCr }
	( .0,+1,-1 ) DUP2 #000f ADD2 /not-head ?{ INCr }
	( +1,-1,-1 ) DUP2 #00ef ADD2 /not-head ?{ INCr }
	( +1, 0,-1 ) DUP2 #00ff ADD2 /not-head ?{ INCr }
	( +1,+1,-1 ) DUP2 #010f ADD2 /not-head ?{ INCr }
	( | z+0 )
	( -1,-1, 0 ) DUP2 #fef0 ADD2 /not-head ?{ INCr }
	( -1, 0, 0 ) DUP2 #ff00 ADD2 /not-head ?{ INCr }
	( -1,+1, 0 ) DUP2 #ff10 ADD2 /not-head ?{ INCr }
	( .0,-1, 0 ) DUP2 #fff0 ADD2 /not-head ?{ INCr }
	( .0,+1, 0 ) DUP2 #0010 ADD2 /not-head ?{ INCr }
	( +1,-1, 0 ) DUP2 #00f0 ADD2 /not-head ?{ INCr }
	( +1, 0, 0 ) DUP2 #0100 ADD2 /not-head ?{ INCr }
	( +1,+1, 0 ) DUP2 #0110 ADD2 /not-head ?{ INCr }
	( | z+1 )
	( -1,-1,+1 ) DUP2 #fef1 ADD2 /not-head ?{ INCr }
	( -1, 0,+1 ) DUP2 #ff01 ADD2 /not-head ?{ INCr }
	( -1,+1,+1 ) DUP2 #ff11 ADD2 /not-head ?{ INCr }
	( .0,-1,+1 ) DUP2 #fff1 ADD2 /not-head ?{ INCr }
	( .0, 0,+1 ) DUP2 #0001 ADD2 /not-head ?{ INCr }
	( .0,+1,+1 ) DUP2 #0011 ADD2 /not-head ?{ INCr }
	( +1,-1,+1 ) DUP2 #00f1 ADD2 /not-head ?{ INCr }
	( +1, 0,+1 ) DUP2 #0101 ADD2 /not-head ?{ INCr }
	( +1,+1,+1 ) DUP2 #0111 ADD2 /not-head ?{ INCr }
