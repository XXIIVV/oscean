@piece-a/handle-char ( c -- )
	DUP #20 GTH ?/<push>
	POP
	( | Parse )
	;&buf LDAk ?{ POP2 JMP2r }
	DUP2 parse-token ,&ptr STR2
	#0000 ;&buf STA2

@walk-comment ( t* -- )
	;&c .File/read DEO2
	;File/succ-lb DEI EQU ?str/<clear>
	[ LIT2 &c $1 "( ] INC NEQ ?walk-comment
	!str/<clear>

@walk-block ( t* -- )
	;&c .File/read DEO2
	;File/succ-lb DEI EQU ?str/<clear>
	[ LIT2 &c $1 "[ ] INC INC NEQ ?walk-block
	reset !str/<clear>

@parse-token ( t* -- )
	LDAk [ LIT "( ] EQU ?walk-comment
	LDAk [ LIT "[ ] EQU ?walk-block
	LDAk [ LIT "% ] NEQ ?{ INC2k macros/<push-str> }
	( blank ) LDAk #00 EQU ?str/<clear>
	( token count ) .stats/tokens LDZ2k INC2 ROT STZ2
	( new scope ) is-new-scope #00 EQU ?{ INC2k scope/<set> }
	( macro reset scope ) LDAk LIT "% NEQ ?{ scope/<unset> }
	( include ) LDA [ LIT "~ ] EQU ?include
	;rules2/end ;rules2
	&>loop2
		( a ) DUP2 ;piece-b/buf scmp-wild STH
		str/cap INC2
		( b ) DUP2 ;piece-a/buf scmp-wild STH
		str/cap INC2 ANDr STHr #00 EQU ?{ LDA2k <print-warn2> }
		INC2 INC2 GTH2k ?/>loop2
	POP2 POP2
	( | 3 elements )
	;rules3/end ;rules3
	&>loop3
		( a ) DUP2 ;piece-c/buf scmp-wild STH
		str/cap INC2
		( b ) DUP2 ;piece-b/buf scmp-wild STH
		str/cap INC2
		( c ) DUP2 ;piece-a/buf scmp-wild STH
		str/cap INC2 ANDr ANDr STHr #00 EQU ?{ LDA2k <print-warn3> }
		INC2 INC2 GTH2k ?/>loop3
	POP2 POP2
	( | tests )
	unsafe-runes/<test>
	mismatch-runes/<test>
	deadzone/<test>
	fallthrough/<test>
	unnamed-buf/<test>
	private/<test>

	;piece-a/buf DUP2 ;piece-b/buf <scpy> !str/<clear>

(
@|helpers )

@is-label ( str* -- flag )
	LDAk LIT "@ EQU ?{
		LDAk LIT "& EQU ?{
			LDAk LIT "} EQU ?{ POP2 #00 JMP2r } } }
	POP2 #01 JMP2r

@is-macro ( str* -- flag )
	LDA LIT "% EQU JMP2r

@is-new-scope ( str* -- str* flag )
	LDAk [ LIT "@ ] NEQ ?/fail
	INC2k LDA chr/is-uc ?/fail
	#01 JMP2r

	&fail ( str* -- flag )
	#00 JMP2r

@is-inside-scope ( str* -- str* flag )
	DUP2 ;scope/buf str/cmp-seg ?{ #00 JMP2r }
	DUP2 scope/length ADD2 LDA LIT "/ EQU JMP2r

@is-jump ( str* -- flag )
	LDAk [ LIT "? ] EQU ?/pass
	&strict ( str* -- flag )
	DUP2 ;brk str/cmp ?/pass
	DUP2 ;jmp str/cmp3 ?/pass
	LDAk [ LIT "! ] EQU ?/pass
	POP2 #00 JMP2r

	&pass ( str* -- flag )
	POP2 #01 JMP2r

@is-litbyte ( str* -- flag )
	LDAk [ LIT ", ] EQU ?/pass
	LDAk [ LIT ". ] EQU ?/pass
	LDAk [ LIT "# ] EQU ?/hex
	POP2 #00 JMP2r

	&hex ( str* -- flag )
	str/len NIP #03 EQU JMP2r

	&pass ( str* -- flag )
	POP2 #01 JMP2r

@is-routine ( str* -- flag )
	LDAk ?{ POP2 #00 JMP2r }
	LDAk chr/is-uc ?{
		LDAk is-runic ?{
			DUP2 macros/exists ?{ POP2 #01 JMP2r } } }
	POP2 #00 JMP2r

@is-litshort ( str* -- flag )
	LDAk [ LIT "; ] EQU ?/pass
	LDAk [ LIT "# ] EQU ?/hex
	POP2 #00 JMP2r

	&hex ( str* -- flag )
	str/len NIP #05 EQU JMP2r

	&pass ( str* -- flag )
	POP2 #01 JMP2r

@is-static ( token* -- flag )
	( ) STH2k is-litshort
	( ) STH2r is-litbyte ORA JMP2r

@is-opcode ( string* -- f )
	LDAk LIT "? EQU ?/on-found
	LDAk LIT "! EQU ?/on-found
	DUP2 ;opcodes/brk str/cmp3 ?/on-found
	,&t STR2
	#2000
	&>l
		#00 OVR #03 MUL ;opcodes ADD2 [ LIT2 &t $2 ] str/cmp3 ?/on-found
		INC GTHk ?/>l
	POP2 #00 JMP2r
	&on-found POP2 #01 JMP2r

@is-ref-rune ( c -- t )
	STH
	;&runes
	&>w
		LDAk STHkr EQU ?{ INC2 LDAk ?/>w }
	LDA STHr EQU JMP2r

	&runes "|$!?.,;_-= $1

@is-runic ( c -- t )
	STH
	;&runes
	&>w
		LDAk STHkr EQU ?{ INC2 LDAk ?/>w }
	LDA STHr EQU JMP2r

	&runes ""|{}%!?@#$%&().,;~ $1 &end

@is-local ( str* -- t )
	LDAk LIT "@ NEQ ?{ POP2 #00 JMP2r }
	LDAk is-ref-rune #00 EQU ?{ INC2 }
	is-inside-scope ?{ POP2 #00 JMP2r }
	;scope/buf !str/cmp-seg

@is-calling-local ( str* -- f )
	LDAk LIT "@ EQU ?{
		&>w
			LDA2k LIT2 "/+ NEQ2 ?{ POP2 #01 JMP2r }
			INC2 LDAk ?/>w }
	POP2 #00 JMP2r

@scmp-wild ( a* b* -- flag )
	OVR2 LDA [ LIT "{ ] EQU ?scmp-template
	OVR2 LDA [ LIT "< ] EQU ?scmp-lut
	!scmp-star

@scmp-lut ( a* b* -- flag )
	SWP2 ,&t STR2
	;dictionaries/end ;dictionaries
	&>l
		LDA2k [ LIT2 &t $2 ] str/cmp ?/dict-match
		#0004 ADD2 GTH2k ?/>l
	,&t LDR2 str/<print>
	#0a18 DEO
	POP2 POP2 POP2 #00 JMP2r

	&dict-match ( target* to* from* -- flag )
	NIP2 INC2 INC2 LDA2 SWP2 afnd INC2 ORA JMP2r

@scmp-template ( a* b* -- flag )
	SWP2 INC2 LDA
	( ) DUP [ LIT "1 ] NEQ ?{ POP !is-litbyte }
	( ) DUP [ LIT "2 ] NEQ ?{ POP !is-litshort }
	( ) DUP [ LIT "% ] NEQ ?{ POP !is-routine }
	POP JMP2r

@scmp-star ( a* b* -- f )
	STH2
	&>l
		LDAk [ LIT "* ] EQU ?/pass
		LDAkr [ LITr "* ] EQUr STHr ?/pass
		LDAk LDAkr STHr ANDk #00 EQU ?/e
		NEQk ?/e
		POP2 INC2 INC2r !/>l

	&e ( -- )
	NIP2 POP2r EQU JMP2r

	&pass ( a* . b* -- f )
	POP2 POP2r #01 JMP2r

	STH2
